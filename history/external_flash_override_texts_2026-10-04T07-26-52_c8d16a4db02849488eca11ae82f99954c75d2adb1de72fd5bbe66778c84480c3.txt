      function byId(id) {
        return document.getElementById(id);
      }

      function detectRegion(hostname, pathname, search) {
        var params = new URLSearchParams(search || "");
        var forced = (params.get("country") || "").toUpperCase();
        if (REGIONS[forced]) {
          return forced;
        }

        var host = (hostname || "").toLowerCase();
        var endings = [
          ["habbo.com.br", "BR"],
          ["habbo.com.tr", "TR"],
          ["habbo.de", "DE"],
          ["habbo.es", "ES"],
          ["habbo.fi", "FI"],
          ["habbo.fr", "FR"],
          ["habbo.it", "IT"],
          ["habbo.nl", "NL"],
          ["habbo.com", "US"]
        ];

        for (var i = 0; i < endings.length; i += 1) {
          var ending = endings[i][0];
          if (host === ending || host.endsWith("." + ending)) {
            return endings[i][1];
          }
        }

        var token = (host + " " + (pathname || "")).match(/(?:^|[._\-\/])(br|de|es|fi|fr|it|nl|tr|us)(?:$|[._\-\/])/i);
        return token ? token[1].toUpperCase() : "US";
      }

      function startCountdown() {
        var n = 60;

      function initTwitter() {

        function tryLoadWidgets() {

          if (window.twttr && window.twttr.widgets && typeof window.twttr.widgets.load === "function") {
            window.twttr.widgets.load(mount);
            return;
